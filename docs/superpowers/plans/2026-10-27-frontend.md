> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace Streamlit with plain HTML/CSS/vanilla-JS pages in `web/` (Indonesian and English, the "Skema graf" look), served next to the API by `python -m autograde serve`, covered by Playwright tests; then delete Streamlit.

**Architecture:** One HTML file per page. Each loads its page module, which builds the page from four shared ES modules (`api.js`, `i18n.js`, `ui.js`, `graph.js`), the shared SPARQL console (`query.js`) and, on the question page, the editor (`editor.js`). The API gains an answer search, more fields on the assessment status, a `lang` parameter on three endpoints and a Content-Security-Policy on the pages. The core gains language-neutral DOT labels, an Indonesian/English text table for the explorer, English column keys for the properties table and the page palette. Browser tests drive Chromium against uvicorn running in a thread on a fresh import of the demo data.

**Tech Stack:** HTML, CSS, vanilla ES modules (no build step); vis-network 9.1.9, @viz-js/viz 3.31.0 (Graphviz in WebAssembly), marked 18.1.1; Python 3.12 (conda env `semantic_web`), FastAPI, pytest, playwright 1.63.0 + pytest-playwright 0.10.0.

## Small refinements to the spec

These came up while mapping the spec onto the code. Each is small and the plan includes it:

- `api.js` retries a 503 only for GET. The spec said "retries", but resending a POST (an import, a correction) could apply it twice.
- Graphviz is vendored from `@viz-js/viz` `dist/viz.js` (the spec named it `viz-standalone.mjs`; 3.31.0 ships one self-contained ES module under that name).
- `GET /results/{run}/{answer}` also returns `assessment_id`, so the answer page can find "the next answer to review".
- The pellet "after" graph labels the rule `(SWRL n)` instead of `(Aturan n)`, the language-neutral form Section 3 of the spec asks for.
- The class flow is split across the Kelas, Penilaian and Jawaban tests: answers imported from a CSV have no triple annotation, so grading them needs an API key. The import is tested on a new course; grading, review, corrections and export on the annotated demo assessment.
- The editor's "invalid" test removes a criterion that a triple still uses. The predicate is a dropdown of valid predicates, so an invalid predicate cannot be chosen.
- In a long-format answer import, the value that marks each question is a text input (default: the question id) with suggestions from the preview rows; the preview returns only five rows.
- `graph.js` keeps the vis-network instance and the details function on the canvas element (`.net`), so tests can toggle and inspect nodes.

```python
def build(ctx, answers, params, human=None, predicates=None, lang="id"):
    """{nodes, edges, details} for one question and any number of graded answers; details in `lang` (id or en)."""
    T = TEXT[lang]
    qid, human = ctx.qid, human or {}
    preds = {p["id"]: p for p in (predicates or io.load_predicates())["predicates"]}
    criteria = {c["id"]: c["description"] for c in ctx.rubric["criteria"]}
    nodes, edges, details = {}, [], {}

    def node(nid, kind, label, title):
        nodes.setdefault(nid, {"id": nid, "kind": kind, "label": label, "title": title})

    def edge(src, dst, kind, label, title=None):
        edges.append({"from": src, "to": dst, "kind": kind, "label": label, "title": title or label})

    # --- the lecturer's key (gold graph) ---
    ref_status = {r.id: [] for r in ctx.refs}
    for r in ctx.refs:
        rid = f"key:{qid}_{r.id}"
        node(rid, "reference", r.id, f"{r.id}: {' / '.join(r.s)} {r.p} {r.o}")
        for s in r.s:
            edge(rid, f"key:{s}", "asserted", "hasSubject")
        edge(rid, f"ag:{r.p}", "asserted", "hasPredicate")
        edge(rid, f"key:{r.o}", "asserted", "hasObject")
    concepts = ctx.gold_raw["concepts"]
    above = gold.ancestors(concepts)
    for c, spec in concepts.items():
        node(f"key:{c}", "concept", c, c)
        for b in (spec or {}).get("broader") or []:
            edge(f"key:{c}", f"key:{b}", "hierarchy", "skos:broader")

    def hierarchy_note(key_c, stud_c):
        if key_c and stud_c and key_c != stud_c:
            if stud_c in above.get(key_c, ()):
                return T["broader_partial"].format(stud=stud_c, key=key_c)
            if key_c in above.get(stud_c, ()):
                return T["narrower_same"].format(stud=stud_c, key=key_c)
        return None

    # --- the graded answers (answer + ledger graphs) ---
    used_preds = {r.p for r in ctx.refs}
    by_triple = {}
    for a in answers:
        aid, short = a["answer_id"], short_answer(qid, a["answer_id"])
        node(f"ans:{aid}", "answer", f"{short} · {a['score']['score']:.0f}", T["answer_title"].format(aid=aid))
        text = dict(a["sentences"])
        matched = {(e.matched_by, e.reference_id): e for e in a["entries"] if e.matched_by}
        for e in a["entries"]:
            ref_status[e.reference_id].append((short, e))
        for sid, sent in a["sentences"]:
            node(f"ans:{sid}", "sentence", sid.rsplit("_", 1)[1], sent)
            details[f"ans:{sid}"] = {"kind": T["sentence_kind"], "title": f"ans:{sid}",
                                    "fields": [[T["text"], sent], [T["answer"], aid]],
                                    "sections": [{"heading": T["from_sentence"],
                                                  "items": [t.id for t in a["students"] if sid in t.sentence_ids] or ["—"]}]}
        for t in a["students"]:
            tid, tn = f"ans:{t.id}", t.id.rsplit("_", 1)[1]
            by_triple[t.id] = t
            used_preds.add(t.p)
            neg = "NOT " if t.negated else ""
            node(tid, "student", f"{short}·{tn}", f"{t.s_text} —{neg}{t.p}→ {t.o_text}")
            edge(tid, f"ans:{aid}", "asserted", "inAnswer")
            edge(tid, f"ag:{t.p}", "asserted", "hasPredicate")
            for role, concept, raw in (("hasSubject", t.s_concept, t.s_text), ("hasObject", t.o_concept, t.o_text)):
                if concept:
                    edge(tid, f"key:{concept}", "asserted", role)
                else:
                    cid = f"ans:{t.id}_{'s' if role == 'hasSubject' else 'o'}"
                    node(cid, "unknown", raw, T["unknown_title"].format(raw=raw))
                    details[cid] = {"kind": T["unknown_kind"], "title": raw,
                                    "fields": [[T["from_triple"], t.id], [T["role"], role]],
                                    "sections": [{"heading": T["why_grey"], "items": [T["why_grey_text"]]}]}
                    edge(tid, cid, "asserted", role)
            for sid in t.sentence_ids:
                edge(tid, f"ans:{sid}", "asserted", "prov:wasDerivedFrom")
            reasoning = []
            for r in ctx.refs:
                sig = a["sigmas"].get((t.id, r.id), 0.0)
                rid, e = f"key:{qid}_{r.id}", matched.get((t.id, r.id))
                if (t.id, r.id) in a["contradictions"]:
                    if t.negated:
                        rule = T["rule2"] if t.p == r.p else T["rule3"]
                        why = T["why_negated"].format(tp=t.p, rp=r.p)
                    else:

"""Helpers for the browser tests: the UI text of a key, grading the demo through the API, console checks."""
import json
import time

import httpx

from autograde import io

HTTP = httpx.Client(trust_env=False, timeout=60)  # never route 127.0.0.1 through a proxy from the environment
I18N = {lang: json.loads((io.ROOT / "web" / "i18n" / f"{lang}.json").read_text(encoding="utf-8")) for lang in ("id", "en")}


def tr(key, lang="id", **values):
    """The text a page shows for a translation key."""
    text = I18N[lang]
    for part in key.split("."):
        text = text[part]
    return text.format(**values) if values else text


def demo_assessment(base):
    course = next(c for c in HTTP.get(f"{base}/api/courses").json() if c["code"] == "DEMO")["id"]
    return HTTP.get(f"{base}/api/courses/{course}/assessments").json()[0]["id"]


def grade_demo(base):
    """Grade the demo assessment's ungraded answers and wait for the job; returns the assessment id."""
    aid = demo_assessment(base)
    r = HTTP.post(f"{base}/api/assessments/{aid}/grading", json={"which": "new"})
    if r.status_code == 422:  # nothing left to grade
        return aid
    job_id, deadline = r.json()["id"], time.monotonic() + 120
    while HTTP.get(f"{base}/api/jobs/{job_id}").json()["status"] == "running":
        if time.monotonic() > deadline:
            raise AssertionError(f"job {job_id} did not finish")
        time.sleep(0.1)
    return aid


def clean(errors, allow=()):
    """The console problems a test collected, minus the ones it provoked on purpose and aborted navigations."""
    return [e for e in errors if "net::ERR_ABORTED" not in e and not any(a in e for a in allow)]

// The JSON API client: one error type, the 503 retry for reads, and URLs for links and downloads.
export class ApiError extends Error {
  constructor(status, body) {
    super(body.message || `HTTP ${status}`);
    this.status = status;
    this.code = body.error || "internal";
    this.details = body.details || [];
  }
}

export function url(path, params = {}) {
  const q = new URLSearchParams();
  for (const [k, v] of Object.entries(params || {})) {
    if (v !== undefined && v !== null && v !== "") q.set(k, v);
  }
  const s = q.toString();
  return `/api${path}${s ? `?${s}` : ""}`;
}

const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

async function request(method, path, { params, json, form, text = false } = {}) {
  const init = { method, headers: {} };
  if (json !== undefined) {
    init.body = JSON.stringify(json);
    init.headers["Content-Type"] = "application/json";
  }
  if (form) init.body = form;
  for (let attempt = 0; ; attempt++) {
    let res;
    try {
      res = await fetch(url(path, params), init);
    } catch {
      throw new ApiError(0, { error: "offline", message: "the server is not running" });
    }
    // Only reads are retried: a resent import or correction could apply twice.
    if (res.status === 503 && method === "GET" && attempt < 3) {
      await sleep(1000 * Number(res.headers.get("Retry-After") || 2));
      continue;
    }
    if (!res.ok) {
      let body;
      try {
        body = await res.json();
      } catch {
        body = { message: res.statusText };
      }
      throw new ApiError(res.status, body);
    }
    return text ? res.text() : res.json();
  }
}

export const api = {
  get: (path, params) => request("GET", path, { params }),
  text: (path, params) => request("GET", path, { params, text: true }),
  post: (path, json, params) => request("POST", path, { json, params }),
  put: (path, json) => request("PUT", path, { json }),
  form: (path, form, params) => request("POST", path, { form, params }),
};

  try {
    current = localStorage.getItem(KEY) === "en" ? "en" : "id";
  } catch {
    current = "id";
  }

  dict = await (await fetch(`/i18n/${current}.json`)).json();

export function setLang(next) {
  try {
    localStorage.setItem(KEY, next);
  } catch {
    return;  // storage blocked: the app stays in Indonesian
  }
  location.reload();
}

export function t(key, vars = {}) {
  const found = key.split(".").reduce((o, k) => (o && typeof o === "object" ? o[k] : undefined), dict);
  let text = found;
  if (typeof found !== "string") {
    if (!("fallback" in vars)) console.warn(`i18n missing: ${key}`);
    text = vars.fallback ?? key;
  }
  return String(text).replace(/\{(\w+)\}/g, (m, k) => (k in vars ? String(vars[k]) : m));
}

  for (const n of root.querySelectorAll("[data-i18n]")) n.textContent = t(n.dataset.i18n);

  for (const n of root.querySelectorAll("[data-i18n-title]")) n.title = t(n.dataset.i18nTitle);

  const n = document.createElement(tag);

export function href(file, params = {}) {
  const q = new URLSearchParams();
  for (const [k, v] of Object.entries(params)) {
    if (v !== undefined && v !== null && v !== "") q.set(k, v);
  }
  const s = q.toString();
  return s ? `${file}?${s}` : file;
}

export function fmt(v) {
  if (v === null || v === undefined || v === "") return "—";
  if (typeof v === "number") return Number.isInteger(v) ? String(v) : v.toFixed(2);
  if (typeof v === "boolean") return v ? "✓" : "—";
  if (Array.isArray(v)) return v.join(", ");
  return String(v);
}

export function tabs(parent, specs) {
  const panels = specs.map(s => el("section", { role: "tabpanel", id: `panel-${s.id}`, "aria-labelledby": `tab-${s.id}`, hidden: true }));
  const rendered = new Set();
  const buttons = specs.map((s, i) => el("button", {
    type: "button", role: "tab", id: `tab-${s.id}`, "aria-controls": `panel-${s.id}`, "aria-selected": "false",
    tabindex: "-1", onclick: () => open(s.id, true),
    onkeydown: e => {
      const step = { ArrowRight: 1, ArrowLeft: -1 }[e.key];
      if (!step) return;
      const next = (i + step + specs.length) % specs.length;
      open(specs[next].id, true);
      buttons[next].focus();
    },
  }, s.label));
  parent.append(el("div", { class: "tabs", role: "tablist" }, buttons), ...panels);

  async function open(id, remember) {
    const i = Math.max(0, specs.findIndex(s => s.id === id));
    if (remember) history.replaceState(null, "", `#${specs[i].id}`);
    buttons.forEach((b, j) => {
      b.setAttribute("aria-selected", String(j === i));
      b.tabIndex = j === i ? 0 : -1;
    });
    panels.forEach((p, j) => { p.hidden = j !== i; });
    if (rendered.has(i)) return;
    rendered.add(i);
    panels[i].replaceChildren(loading());
    try {
      await specs[i].render(panels[i]);
    } catch (e) {
      panels[i].replaceChildren(errorPanel(e));
      report(e);
    }
  }
  return open(location.hash.slice(1), false);
}

// Graph drawing: DOT through Graphviz (WebAssembly, loaded on first use) and JSON graphs through vis-network.
import { t } from "./i18n.js";
import { el, fill } from "./ui.js";

let viz = null;
let vis = null;

export async function dot(container, text) {
  viz ??= import("/vendor/viz.js").then(m => m.instance());
  const svg = (await viz).renderSVGElement(text);
  svg.removeAttribute("width");  // scale with the panel; the viewBox keeps the proportions
  svg.removeAttribute("height");
  container.replaceChildren(el("div", { class: "graph-panel dot-panel" }, svg));

}

function loadVis() {
  vis ??= new Promise((resolve, reject) => {
    const s = el("script", { src: "/vendor/vis-network.min.js" });
    s.onload = () => resolve(window.vis);
    s.onerror = reject;
    document.head.append(s);
  });
  return vis;
}

function details(target, graph, id) {

function legendButton(label, swatch, shown, onToggle) {
  const b = el("button", { type: "button", "aria-pressed": String(shown) }, swatch, label);
  b.addEventListener("click", () => {
    const on = b.getAttribute("aria-pressed") !== "true";

export async function network(container, graph, nodeTypes, edgeTypes, { height = 600 } = {}) {
  const V = await loadVis();
  const hiddenKinds = new Set(Object.entries(nodeTypes).filter(([, v]) => !v[3]).map(([k]) => k));
  const hiddenEdges = new Set(Object.entries(edgeTypes).filter(([, v]) => !v[4]).map(([k]) => k));
  const nodes = new V.DataSet(graph.nodes.map(n => {
    const [, color, shape] = nodeTypes[n.kind];
    const box = shape === "box" || shape === "ellipse";
    return { id: n.id, label: n.label, title: n.title, kind: n.kind, shape,
      color: { background: color, border: color, highlight: { background: color, border: "#16202e" } },
      font: box ? { color: "#fff", size: 13 } : { color: "#16202e", size: 13 },
      size: n.kind === "answer" ? 22 : n.kind === "student" ? 12 : 14, hidden: hiddenKinds.has(n.kind) };
  }));
  const edges = new V.DataSet(graph.edges.map((e, i) => {
    const [, color, dashed, width] = edgeTypes[e.kind];
    return { id: `e${i}`, from: e.from, to: e.to, kind: e.kind, label: e.label, fullLabel: e.label, title: e.title,
      arrows: e.kind === "ontology" ? "" : "to", dashes: dashed, width,
      color: { color, highlight: "#16202e", opacity: e.kind === "asserted" ? 0.7 : 1 },
      font: { size: 9, color: "#5b6472", strokeWidth: 2, align: "top" }, hidden: hiddenEdges.has(e.kind) };
  }));
  const canvas = el("div", { class: "net", style: { height: `${height}px` } });
  const side = el("aside", { class: "details" }, el("p", { class: "muted" }, t("graph.hint")));
  const legend = el("div", { class: "legend" },
    Object.entries(nodeTypes).map(([k, v]) => legendButton(t(`node.${k}`, { fallback: v[0] }),
      el("span", { class: "swatch", style: { background: v[1] } }), !hiddenKinds.has(k),
      on => nodes.update(nodes.get({ filter: n => n.kind === k }).map(n => ({ id: n.id, hidden: !on }))))),
    Object.entries(edgeTypes).map(([k, v]) => legendButton(t(`edge.${k}`, { fallback: v[0] }),
      el("span", { class: v[2] ? "swatch line dash" : "swatch line", style: { borderTopColor: v[1] } }),
      !hiddenEdges.has(k),
      on => edges.update(edges.get({ filter: x => x.kind === k }).map(x => ({ id: x.id, hidden: !on }))))));
  const search = el("input", { type: "search", placeholder: t("graph.search"), "aria-label": t("graph.search") });
  const fit = el("button", { type: "button", class: "btn secondary" }, t("graph.fit"));
  const freeze = el("button", { type: "button", class: "btn secondary" }, t("graph.freeze"));
  const labels = el("input", { type: "checkbox", checked: true });
  fill(container, legend,
    el("div", { class: "toolbar" }, search, fit, freeze, el("label", {}, labels, " ", t("graph.labels")),
      el("span", { class: "muted small" }, t("graph.counts", { nodes: graph.nodes.length, edges: graph.edges.length }))),
    graph.truncated ? el("p", { class: "muted" }, t("graph.truncated", { n: graph.edges.length })) : null,
    el("div", { class: "graph-body" }, canvas, side));
  const net = new V.Network(canvas, { nodes, edges }, {
    physics: { solver: "forceAtlas2Based", forceAtlas2Based: { gravitationalConstant: -60, springLength: 110 },
      stabilization: { iterations: 250 } },
    interaction: { hover: true, tooltipDelay: 120, navigationButtons: false },
    edges: { smooth: { type: "continuous" } },
  });
  const show = id => details(side, graph, id);
  net.on("click", e => { if (e.nodes.length) show(e.nodes[0]); });
  fit.onclick = () => net.fit({ animation: true });
  let frozen = false;
  freeze.onclick = () => {
    frozen = !frozen;
    net.setOptions({ physics: { enabled: !frozen } });
    freeze.textContent = t(frozen ? "graph.unfreeze" : "graph.freeze");
  };
  labels.onchange = () => edges.update(edges.get().map(x => ({ id: x.id, label: labels.checked ? x.fullLabel : " " })));
  search.oninput = () => {
    const q = search.value.trim().toLowerCase();
    if (q.length < 2) return;
    const hit = nodes.get({ filter: n => !n.hidden && (n.label.toLowerCase().includes(q) || (n.title || "").toLowerCase().includes(q)) })[0];
    if (hit) {
      net.selectNodes([hit.id]);
      net.focus(hit.id, { scale: 1.2, animation: true });
      show(hit.id);
    }
  };
  canvas.network = net;  // for tests and the browser console
  canvas.showDetails = show;
  return net;
}

// The SPARQL console: a template picker, an editor (Ctrl+Enter runs it) and the result as a table or Turtle.
import { api } from "./api.js";
import { lang, t } from "./i18n.js";
import { button, code, el, field, iri, loading, select, table } from "./ui.js";

export async function sparqlConsole(container, { graphs = null, scope = "kg", template = null } = {}) {
  const templates = await api.get("/sparql/templates", { scope, lang: lang() });
  const names = Object.keys(templates);
  const first = template && templates[template] ? template : names[0];
  const picker = select(names.map(n => [n, n]), first);
  const editor = el("textarea", { class: "code", spellcheck: "false", "aria-label": t("sparql.query"),
    value: templates[first] ?? "" });
  picker.addEventListener("change", () => { editor.value = templates[picker.value]; });
  const out = el("div", { class: "results", "aria-live": "polite" });
  const run = button(t("sparql.run"), async () => {
    out.replaceChildren(loading());
    try {
      const res = await api.post("/sparql", { query: editor.value, graphs });
      out.replaceChildren(...(res.kind === "turtle"
        ? [code(res.turtle || t("sparql.empty_graph"))]
        : [table(res, { translate: false }), el("p", { class: "small muted" }, t("sparql.rows", { n: res.rows.length }))]));
    } catch (e) {
      out.replaceChildren();
      throw e;
    }
  }, { kind: "primary", slot: out });
  editor.addEventListener("keydown", e => {
    if (e.key === "Enter" && (e.ctrlKey || e.metaKey)) {
      e.preventDefault();
      run.click();
    }
  });
  container.replaceChildren(
    el("p", { class: "small muted" }, graphs ? [t("sparql.over_graphs"), " ", graphs.map(g => [iri(g), " "])] : t("sparql.over_store")),
    field(t("sparql.template"), picker), editor,
    el("div", { class: "actions" }, run, el("span", { class: "small muted" }, t("sparql.shortcut"))), out);
}
```

// Jawaban: one graded answer: score and SPARQL check, ledger, graphs, reasoning, RDF, SPARQL, OWL, and its review.
import { api, url } from "./api.js";
import { dot, network } from "./graph.js";
import { lang, t } from "./i18n.js";
import { sparqlConsole } from "./query.js";
import { button, code, dl, el, fail, fill, fmt, href, iri, ledgerChip, panel, param, records, requireReviewer,
  score, select, start, stat, statusChip, table, tabs, toast } from "./ui.js";

const enc = encodeURIComponent;
const LEDGER_STATUSES = ["MATCHED", "PARTIAL", "MISSING", "CONTRADICTED"];

const statusOf = r => (r.stale ? "usang" : r.reviewed ? "ditinjau" : "dinilai");

function sparqlCheck(r) {
  const c = r.recompute;
  return el("details", { class: "panel" },
    el("summary", {}, el("span", { class: c.ok ? "check-ok" : "check-bad" }, c.ok ? "✓ " : "✗ "),
      t("jawaban.check", { python: score(c.python), sparql: score(c.sparql) })),
    el("p", { class: "small muted" }, t("jawaban.check_graphs"), " ", c.graphs.map(g => [iri(g), " "])),
    code(c.query),
    table({ columns: ["answer", "raw"], rows: c.rows.map(x => [x.answer, x.raw]) }));
}

function header(r, run, answer) {
  const sc = r.score;
  return el("section", { class: "panel" },
    el("div", { class: "row" }, el("h1", {}, iri(`ans:${answer}`)), el("span", { id: "answer-status" }, statusChip(statusOf(r)))),
    el("p", { class: "muted" }, t("jawaban.meta", { question: r.question_id, run, graded: r.graded || "—" }), " · ",
      el("a", { href: href("soal.html", { id: r.question_id }) }, t("jawaban.open_question")),
      r.reviewed_by ? ` · ${t("jawaban.reviewed_by", { by: r.reviewed_by })}` : ""),
    el("div", { class: "row" },
      el("div", {}, el("div", { class: "score-big" }, score(sc.score)), el("div", { class: "small muted" }, t("jawaban.system_score"))),
      r.score_corrected ? el("div", {}, el("div", { class: "score-big" }, score(r.score_corrected.score)),
        el("div", { class: "small muted" }, t("jawaban.corrected_score"))) : null,
      el("div", { class: "stats" }, stat("Σ w·m", fmt(sc.sum_weighted_credit)), stat(t("jawaban.penalty"), fmt(sc.sum_penalty)),
        stat("Σ w", fmt(sc.sum_weights)))),
    sparqlCheck(r),
    el("div", { class: "actions" }, dl(url("/export/owl", { run, answer }), t("jawaban.owl"))));
}

function ledgerTable(entries, sentences, controls = null) {
  const text = Object.fromEntries(sentences);
  const byRef = Object.fromEntries(entries.map(e => [e.reference_id, e]));
  const columns = ["ref", "reference", "criterion", "weight", "status", "credit", "sigma", "evidence", "override"];
  if (controls) columns.push("correct_to", "reason");
  const rows = entries.map(e => columns.map(c => ({
    ref: e.reference_id, reference: e.reference ?? t("jawaban.not_in_key"), criterion: e.criterion, weight: e.weight,
    status: e.status, credit: e.credit, sigma: e.sigma,
    evidence: e.sentence_ids.map(s => text[s]).filter(Boolean).join(" ") || null,
    override: e.override ? `${e.override.by}: ${e.override.reason}` : null, correct_to: null, reason: null,
  })[c]));
  return table({ columns, rows }, { cell: (c, v, row) => {
    const e = byRef[row[0]];
    if (c === "ref") return el("code", {}, v);
    if (c === "status") return [ledgerChip(v), e.needs_review ? " ⚑" : ""];
    if (c === "correct_to") return controls[e.reference_id].select;
    if (c === "reason") return controls[e.reference_id].reason;
    return undefined;
  } });
}

function reviewControls(entries) {
  return Object.fromEntries(entries.map(e => [e.reference_id, {
    entry: e,
    select: select(LEDGER_STATUSES.map(s => [s, t(`ledger.${s}`)]), e.status,
      { "aria-label": t("review.status_for", { ref: e.reference_id }) }),
    reason: el("input", { "aria-label": t("review.reason_for", { ref: e.reference_id }) }),
  }]));
}

async function nextToReview(assessmentId, current) {
  const st = await api.get(`/assessments/${assessmentId}/status`);
  const col = name => st.columns.indexOf(name);
  const pending = st.rows.filter(row => row[col("status")] === "dinilai" && row[col("answer_id")] !== current)
    .sort((a, b) => (b[col("flagged")] - a[col("flagged")]) || (b[col("contradictions")] - a[col("contradictions")]));
  return pending.length ? pending[0][col("answer_id")] : null;
}

async function reviewBlock(r, run, answer, refresh, controls) {
  if (!r.assessment_id) return null;
  if (r.stale) {
    return el("div", { class: "notice" }, t("review.stale"), " ",
      el("a", { href: href("penilaian.html", { id: r.assessment_id }) }, t("review.to_assessment")));
  }
  const next = await nextToReview(r.assessment_id, answer);
  const nextLink = next ? el("a", { href: href("jawaban.html", { run, answer: next }) }, t("review.next"))
    : el("span", { class: "muted" }, t("review.none_left"));
  if (r.reviewed) return el("div", { class: "actions" }, el("span", {}, t("review.done", { by: r.reviewed_by })), nextLink);
  const changed = () => Object.values(controls).filter(c => c.select.value !== c.entry.status);
  const slot = el("div");
  const save = button(t("review.save"), async () => {
    if (!changed().length) throw fail("invalid", t("review.nothing_changed"));
    const by = requireReviewer();
    const res = await api.post(`/results/${enc(run)}/${enc(answer)}/corrections`, { by,
      corrections: changed().map(c => ({ reference_id: c.entry.reference_id, status: c.select.value, reason: c.reason.value })) });
    toast(t("review.saved", { score: score(res.score.score), sparql: score(res.score.sparql_score) }));
    await refresh();
  }, { slot });
  const mark = button(t("review.mark"), async () => {
    if (changed().length) throw fail("invalid", t("review.save_first"));
    const by = requireReviewer();
    await api.post(`/results/${enc(run)}/${enc(answer)}/review`, { by });
    toast(t("review.marked", { answer }));
    await refresh();
  }, { kind: "primary", slot });
  return el("section", { class: "panel" }, el("h2", {}, t("review.title")), el("p", { class: "small muted" }, t("review.help")),
    el("div", { class: "actions" }, save, mark, nextLink), slot);
}

async function ledgerTab(p, r, run, answer, refresh) {
  const reviewable = Boolean(r.assessment_id) && !r.stale && !r.reviewed;
  const controls = reviewable ? reviewControls(r.ledger_corrected || r.ledger) : null;
  const fb = r.feedback;
  fill(p,
    await reviewBlock(r, run, answer, refresh, controls),
    el("h2", {}, t("jawaban.ledger")), ledgerTable(r.ledger, r.sentences, r.ledger_corrected ? null : controls),
    r.ledger_corrected ? [el("h2", {}, t("jawaban.ledger_corrected")), ledgerTable(r.ledger_corrected, r.sentences, controls)] : null,
    panel(t("jawaban.feedback"),
      el("ul", {}, fb.claims.map(c => el("li", {}, c.text, " ", el("small", { class: "muted" }, `(${c.ledger_entry_ids.join(", ")})`)))),
      el("p", {}, el("em", {}, fb.summary)),
      el("p", { class: "small muted" }, t(fb.source === "llm" ? "jawaban.feedback_llm" : "jawaban.feedback_template"))),
    panel(t("jawaban.sentences"), el("ol", {}, r.sentences.map(([id, s]) => el("li", {}, iri(`ans:${id}`), " ", s)))));
}

async function reasoningTab(p, r, base) {

  const prm = v.params;
  const q = r.question_id;
  const before = el("div");
  const after = el("div");
  fill(p,
    el("h2", {}, t("jawaban.step_normalise")),

    table(records(v.students.map(s => ({ triple: s.id, subject: s.s_text, s_concept: s.s_concept ?? "? (embedding)", predicate: s.p,
      negated: s.negated, object: s.o_text, o_concept: s.o_concept ?? "? (embedding)", sentences: s.sentence_ids.join(", ") })),
    ["triple", "subject", "s_concept", "predicate", "negated", "object", "o_concept", "sentences"])),

      `ans:${s} ag:alignedSubject key:${q}_${ref} ; ag:alignedObject key:${q}_${ref} .`).join("\n")) : null,

  if (v.pellet) await Promise.all([dot(before, v.pellet.before), dot(after, v.pellet.after)]);
}

// Penilaian: an assessment's status, the background grading job (followed after a reload), bulk review, exports.
import { api, url } from "./api.js";
import { t } from "./i18n.js";
import { button, dl, el, errorPanel, fail, href, iri, panel, param, report, requireReviewer, score, start,
  statusChip, table, toast } from "./ui.js";

const STATUSES = ["belum dinilai", "usang", "dinilai", "ditinjau"];
const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));

function progressView(job, cancel) {
  return el("div", {},
    el("progress", { max: job.total, value: job.done }),
    el("p", {}, t("penilaian.progress", { done: job.done, total: job.total }), job.current ? ` · ${job.current}` : ""),
    job.status === "running" ? cancel : el("p", {}, t(`penilaian.job_${job.status}`)),
    job.failures.length ? el("details", { open: true }, el("summary", {}, t("penilaian.failures", { n: job.failures.length })),
      el("ul", {}, job.failures.map(f => el("li", {}, `${f.answer_id}: ${f.error}`)))) : null);
}

async function follow(job, area, controls, refresh) {
  controls.hidden = true;  // one job at a time: no start buttons while this one runs
  const cancel = button(t("penilaian.cancel"), async () => { job = await api.post(`/jobs/${job.id}/cancel`); },
    { kind: "danger", slot: area });
  while (job.status === "running") {
    area.replaceChildren(progressView(job, cancel));
    await sleep(1000);
    if (!area.isConnected) return;  // the page was rendered again meanwhile
    job = await api.get(`/jobs/${job.id}`);
  }
  await refresh();
}

function exportsPanel(id, st) {
  const locked = !st.ready;
  const missing = Object.entries(st.counts).filter(([s]) => s !== "ditinjau")
    .map(([s, n]) => `${n} ${t(`status.${s}`)}`).join(", ");
  const final = kind => (locked ? el("button", { type: "button", class: "btn secondary", disabled: true }, t(`penilaian.final_${kind}`))
    : dl(url(`/assessments/${id}/exports/${kind}.csv`), t(`penilaian.final_${kind}`)));
  const draft = kind => dl(url(`/assessments/${id}/exports/${kind}.csv`, { draft: true }), t(`penilaian.draft_${kind}`));
  return panel(t("penilaian.exports"),
    el("div", { class: "actions" }, final("grades"), final("feedback"),
      locked ? el("span", { class: "reason" }, t("penilaian.locked", { missing })) : null),
    el("div", { class: "actions" }, draft("grades"), draft("feedback")),
    el("p", { class: "small muted" }, t("penilaian.mean_note")));
}

start("penilaian.html", "penilaian.title", async (main, refresh) => {
  const id = param("id");
  if (!id) throw fail("not_found", t("penilaian.missing"));
  const st = await api.get(`/assessments/${encodeURIComponent(id)}/status`);
  const col = name => st.columns.indexOf(name);
  const head = el("section", { class: "panel" },
    el("h1", {}, `${st.course.code} · ${st.assessment.name}`),
    el("p", { class: "muted" }, "run ", iri(st.run_id), " · ",
      el("a", { href: href("kelas.html", { course: st.course.id }) }, t("penilaian.course"))),
    el("div", { class: "chips counts" }, STATUSES.filter(s => st.counts[s])
      .map(s => el("span", { class: `chip st-${s.replace(" ", "-")}` }, `${st.counts[s]} ${t(`status.${s}`)}`))),
    st.synthetic ? el("p", { class: "small muted" }, t("penilaian.synthetic")) : null);
  if (!st.rows.length) {
    main.replaceChildren(head, el("p", { class: "muted" }, t("penilaian.no_answers")));
    return;
  }
  const area = el("div", { id: "job-area", "aria-live": "polite" });
  const llm = el("input", { type: "checkbox" });
  const startJob = (which, status) => button(t(`penilaian.grade_${which}`, { n: st.counts[status] || 0 }), async () => {
    const job = await api.post(`/assessments/${id}/grading`, { which, llm_feedback: llm.checked });
    follow(job, area, controls, refresh).catch(e => { area.append(errorPanel(e)); report(e); });
  }, { kind: which === "new" ? "primary" : "secondary", disabled: !st.counts[status], slot: area });
  const controls = el("div", {}, el("label", {}, llm, " ", t("penilaian.llm_feedback")),
    el("div", { class: "actions" }, startJob("new", "belum dinilai"), startJob("stale", "usang")));
  const job = st.job;
  if (job && job.status !== "running") {
    area.append(el("p", { class: "small muted" }, t("penilaian.last_job", { id: job.id })), progressView(job, null));
  }
  const approve = button(t("penilaian.approve", { n: st.approvable.length }), async () => {
    const res = await api.post(`/assessments/${id}/review/bulk`, { by: requireReviewer() });
    toast(t("penilaian.approved", { n: res.reviewed.length }));
    await refresh();
  }, { kind: "primary", disabled: !st.approvable.length });
  main.replaceChildren(head,
    panel(t("penilaian.grading"), el("p", { class: "small muted" }, t("penilaian.grading_help")), controls, area),
    panel(t("penilaian.table"), table(st, { cell: (c, v, row) => {
      if (c === "status") return statusChip(v);
      if (c === "answer_id") {
        return row[col("status")] === "belum dinilai" ? v : el("a", { href: href("jawaban.html", { run: st.run_id, answer: v }) }, v);
      }
      if (c === "score" || c === "final") return score(v);
      if (c === "flagged") return v ? `⚑ ${v}` : "—";
      return undefined;
    } })),
    panel(t("penilaian.review"), el("p", { class: "small muted" }, t("penilaian.review_help")), el("div", { class: "actions" }, approve)),
    exportsPanel(id, st));
  if (job && job.status === "running") {
    follow(job, area, controls, refresh).catch(e => { area.append(errorPanel(e)); report(e); });
  }
});

// Kelas: courses and assessments, students, question upload, and the LMS CSV import through a column mapping.
import { api } from "./api.js";
import { t } from "./i18n.js";
import { add, button, el, errorPanel, fail, field, fill, href, loading, panel, param, records, report, select, start,
  table, tabs, toast } from "./ui.js";

const MAX_UPLOAD = 10 * 1024 * 1024;
const NONE = [["", "—"]];

function upload(name, files) {
  const fd = new FormData();
  for (const f of files) {
    if (f.size > MAX_UPLOAD) throw fail("too_large", t("kelas.too_large", { name: f.name }));
    fd.append(name, f);
  }
  return fd;
}

// The file input, its preview, then the mapping form once the columns are known; send(path, mapping, dryRun) posts
// the file again with the mapping (the import is stateless).
function csvStep(p, before, drawMapping) {
  const file = el("input", { type: "file", accept: ".csv,text/csv" });
  const mapping = el("div");
  const result = el("div", { "aria-live": "polite" });
  file.addEventListener("change", async () => {
    mapping.replaceChildren(loading());
    result.replaceChildren();
    try {
      const pv = await api.form("/imports/preview", upload("file", [file.files[0]]));
      mapping.replaceChildren(el("p", { class: "small muted" }, t("import.preview", { n: pv.count })), table(pv, { translate: false }));
      await drawMapping(mapping, pv, async (path, map, dryRun) => {
        const fd = upload("file", [file.files[0]]);
        fd.append("mapping", JSON.stringify(map));
        return api.form(path, fd, { dry_run: dryRun });
      }, result);
    } catch (e) {
      mapping.replaceChildren(errorPanel(e));
      report(e);
    }
  });
  p.replaceChildren(before, field(t("import.file"), file), mapping, result);
}

function answersTab(p, courseId, assessments, refresh) {
  const usable = assessments.filter(a => a.questions.length);
  if (!usable.length) {
    p.replaceChildren(el("p", { class: "muted" }, t("import.no_questions")));
    return;
  }
  const which = select(usable.map(a => [String(a.id), a.name]), String(usable[0].id));
  csvStep(p, field(t("import.assessment"), which), (box, pv, send, result) => {
    const cols = pv.columns.map(c => [c, c]);
    const questions = usable.find(a => String(a.id) === which.value).questions;
    const idCol = select(cols, pv.columns[0]);
    const names = pv.columns.map(c => el("label", {}, el("input", { type: "checkbox", value: c }), " ", c));
    const layout = select([["wide", t("import.wide")], ["long", t("import.long")]], "wide");
    const wide = Object.fromEntries(questions.map(q => [q, select([...NONE, ...cols], pv.columns.includes(q) ? q : "")]));
    const qCol = select(cols, pv.columns[0]);
    const aCol = select(cols, pv.columns[pv.columns.length - 1]);
    const values = el("datalist", { id: "question-values" });
    const long = Object.fromEntries(questions.map(q => [q, el("input", { value: q, list: "question-values" })]));
    const split = select([["test", t("import.split_test")], ["pilot", t("import.split_pilot")]], "test");
    const wideBox = el("div", {}, questions.map(q => field(t("import.answer_col", { q }), wide[q])));
    const longBox = el("div", { hidden: true }, el("div", { class: "row" }, field(t("import.question_col"), qCol),
      field(t("import.answers_col"), aCol)), values, questions.map(q => field(t("import.value_for", { q }), long[q])));
    const fillValues = () => values.replaceChildren(...[...new Set(pv.rows.map(r => r[pv.columns.indexOf(qCol.value)]))]
      .map(v => el("option", { value: v })));
    qCol.addEventListener("change", fillValues);
    fillValues();
    layout.addEventListener("change", () => {
      wideBox.hidden = layout.value !== "wide";
      longBox.hidden = layout.value === "wide";
    });
    const mapping = () => {
      const columns = {};
      for (const q of questions) {
        if (layout.value === "wide" && wide[q].value) columns[q] = { column: wide[q].value };
        if (layout.value === "long" && long[q].value.trim()) {
          columns[q] = { column: aCol.value, filter_column: qCol.value, filter_value: long[q].value.trim() };
        }
      }
      return { id_col: idCol.value, name_cols: names.map(l => l.firstChild).filter(i => i.checked).map(i => i.value),
        columns, split: split.value };
    };
    const go = dryRun => async () => {
      const r = await send(`/courses/${courseId}/answers`, mapping(), dryRun);
      if (!dryRun) {
        toast(t("import.answers_done", { n: r.added }));
        await refresh();
        return;
      }
      fill(result, el("p", {}, t("import.check_result", { new: r.new, blank: r.blank, existing: r.existing.length })),
        r.existing.length ? el("details", {}, el("summary", {}, t("import.existing")), el("p", { class: "small" }, r.existing.join(", "))) : null,
        table(records(r.answers, ["answer_id", "external_id", "name", "text"])));
    };
    box.append(field(t("import.id_col"), idCol), el("fieldset", {}, el("legend", {}, t("import.name_cols")), el("div", { class: "row" }, names)),
      field(t("import.layout"), layout), wideBox, longBox, field(t("import.split"), split),
      el("div", { class: "actions" }, button(t("import.check"), go(true), { slot: result }),
        button(t("import.run"), go(false), { kind: "primary", slot: result })));
  });
}

// Soal: the question list; one question's key with its graphs and Turtle, its answers, and grading one now.
import { api, url } from "./api.js";
import { dot } from "./graph.js";
import { t } from "./i18n.js";
import { button, code, dl, el, field, href, iri, loading, panel, param, records, score, select, start, stripComments,
  table, tabs, triple } from "./ui.js";

const enc = encodeURIComponent;

async function list(main) {
  const qs = await api.get("/questions");
  main.replaceChildren(el("div", { class: "row" }, el("h1", {}, t("soal.title"))),
    table(records(qs, ["id", "prompt", "language", "assessment_id", "answers", "results"]), { cell: (c, v) => {
      if (c === "id") return el("a", { href: href("soal.html", { id: v }) }, v);
      if (c === "prompt") {
        const s = stripComments(v);
        return s.length > 120 ? `${s.slice(0, 120)}…` : s;
      }
      return undefined;
    } }));
}

function notice(g) {
  if (g.drafted_by) return el("div", { class: "notice" }, t("soal.drafted", { by: g.drafted_by }));
  if (!g.validated_by) return el("div", { class: "notice" }, t("soal.not_validated"));
  return el("p", { class: "small muted" }, t("soal.validated", { by: g.validated_by }));
}

async function keyTab(p, qid, g) {
  const graph = el("div");
  const view = select([["concept", t("soal.concept_graph")], ["reified", t("soal.reified_graph")]], "concept");
  view.addEventListener("change", () => dot(graph, view.value === "concept" ? g.concept_dot : g.reified_dot));
  p.replaceChildren(
    el("div", { class: "grid2" }, panel(t("soal.prompt"), el("p", {}, stripComments(g.prompt))),
      panel(t("soal.reference"), el("p", {}, stripComments(g.reference_answer)))),
    panel(t("soal.criteria"), table(records(g.criteria, ["id", "description", "weight"]))),
    panel(t("soal.triples"), table(records(g.triples.map(r => ({ ...r, triple: r })), ["id", "triple", "criterion", "weight"]),
      { cell: (c, v) => (c === "triple" ? triple({ s: v.s.join(" / "), p: v.p, o: v.o, negated: false }) : undefined) })),
    panel(t("soal.concepts"), el("p", { class: "small muted" }, t("soal.concepts_help")),
      table(records(g.concepts, ["concept", "aliases", "broader"]), { cell: (c, v) => (c === "concept" ? iri(`key:${v}`) : undefined) })),
    panel(t("soal.graph"), el("p", { class: "small muted" }, t("soal.graph_help"), " ", iri(g.graph)),
      field(t("soal.graph_view"), view), graph),
    panel("Turtle", el("div", { class: "actions" }, dl(url("/graphs/turtle", { name: g.graph }), t("common.download")),
      dl(url(`/questions/${enc(qid)}/files.zip`), t("soal.zip"))), code(g.turtle)));
  await dot(graph, g.concept_dot);
}

function tryResult(res) {
  return panel(null,
    el("div", { class: "row" }, iri(`ans:${res.answer_id}`), el("span", { class: "score-big" }, score(res.score.score)),
      el("span", { class: res.recompute.ok ? "check-ok" : "check-bad" }, `${res.recompute.ok ? "✓" : "✗"} SPARQL ${score(res.score.sparql_score)}`),
      el("span", {}, t("soal.baseline", { score: score(res.baseline_b1) }))),
    el("a", { href: href("jawaban.html", { run: res.run_id, answer: res.answer_id }) }, t("soal.open_result")));
}

async function answersTab(p, qid) {
  const data = await api.get(`/questions/${enc(qid)}/answers`);
  const col = name => data.columns.indexOf(name);

    result.replaceChildren(loading());
    try {
      result.replaceChildren(tryResult(await api.post(`/questions/${enc(qid)}/try`, { ...body, llm_feedback: llm.checked })));
    } catch (e) {
      result.replaceChildren();
      throw e;
    }
  };
  const pasted = el("textarea", { rows: 4 });
  const rows = { columns: [...data.columns, "try"], rows: data.rows.map(r => [...r, null]) };
  p.replaceChildren(
    el("p", { class: "small muted" }, t("soal.try_help")), el("label", {}, llm, " ", t("soal.llm_feedback")), result,
    table(rows, { cell: (c, v, row) => {
      if (c === "try") {
        return button(t("soal.try"), grade({ answer_id: row[col("answer_id")], mode: row[col("annotated")] ? "gold" : "llm" }),
          { slot: result });
      }
      if (c === "human") return score(v);
      return undefined;
    } }),
    panel(t("soal.paste_title"), field(t("soal.paste"), pasted),
      el("div", { class: "actions" }, button(t("soal.try_pasted"), () => grade({ text: pasted.value, mode: "llm" })(),
        { kind: "primary", slot: result }))));
}

// The question editor: the key's tables as editable rows, a live preview (the errors or the gold graph), saving.
import { api } from "./api.js";
import { dot } from "./graph.js";
import { t } from "./i18n.js";
import { button, code, el, errorPanel, field, fill, iri, panel, report, select, toast } from "./ui.js";

export const EMPTY_DRAFT = { prompt: "", reference_answer: "", language: "id", validated_by: "",
  criteria: [{ id: "K1", description: "" }], concepts: [], triples: [] };

const split = value => value.split(",").map(s => s.trim()).filter(Boolean);

// One editable table over `items` (the draft's rows, edited in place). kind: text, list (comma-separated), number, select.
function rowsEditor(spec, items, onChange) {
  const body = el("tbody");
  const control = (s, item, i) => {
    const label = `${s.label} ${i + 1}`;
    if (s.kind === "select") {
      const options = s.options();
      if (item[s.key] && !options.some(([v]) => v === item[s.key])) {
        options.push([item[s.key], `${item[s.key]} (${t("editor.unknown")})`]);
      }
      return select([["", "—"], ...options], item[s.key] ?? "", { "aria-label": label,
        onchange: e => { item[s.key] = e.target.value; onChange(); } });
    }
    const value = s.kind === "list" ? (item[s.key] || []).join(", ") : item[s.key] ?? "";
    return el("input", { type: s.kind === "number" ? "number" : "text", step: s.kind === "number" ? "0.5" : null,
      min: s.kind === "number" ? "0.1" : null, list: s.datalist || null, value: String(value), "aria-label": label,
      oninput: e => {
        const v = e.target.value;
        item[s.key] = s.kind === "list" ? split(v) : s.kind === "number" ? (v === "" ? null : Number(v)) : v;
        onChange();
      } });
  };
  const render = () => body.replaceChildren(...items.map((item, i) => el("tr", {},
    spec.map(s => el("td", {}, control(s, item, i))),
    el("td", {}, el("button", { type: "button", class: "btn danger", "aria-label": t("editor.remove_row", { n: i + 1 }),
      onclick: () => { items.splice(i, 1); render(); onChange(); } }, "×")))));
  render();
  const element = el("div", {},
    el("div", { class: "table-wrap" }, el("table", {}, el("thead", {}, el("tr", {}, spec.map(s => el("th", {}, s.label)),
      el("th", {}))), body)),
    el("button", { type: "button", class: "btn secondary", onclick: () => { items.push({}); render(); onChange(); } },
      t("editor.add_row")));
  return { element, render };
}

export async function editor(container, { qid = null, draft, baseId = null, assessments = null, onSaved }) {
  const d = structuredClone(draft);
  let predicates = await api.get("/predicates", { lang: d.language });
  const concepts = () => d.concepts.map(c => (c.concept || "").trim()).filter(Boolean);
  const criteria = () => d.criteria.map(c => (c.id || "").trim()).filter(Boolean);
  const datalist = el("datalist", { id: "concepts-list" });
  const status = el("p", { class: "small muted", "aria-live": "polite" });
  const preview = el("div");
  const newId = qid ? null : el("input", { placeholder: "Q2" });
  const target = assessments ? select(assessments.map(a => [String(a.id), a.label]), String(assessments[0].id)) : null;
  let seq = 0;
  let timer = null;

  const save = button(t("editor.save"), async () => {
    if (qid) {
      const r = await api.put(`/questions/${encodeURIComponent(qid)}`, d);
      toast(t("editor.saved", { id: qid, n: r.stale_results }));
      await onSaved(qid);
    } else {
      const r = await api.post("/questions", { id: newId.value.trim(), assessment_id: Number(target.value), draft: d });
      await onSaved(r.id);
    }
  }, { kind: "primary", disabled: true });

  async function check() {
    const mine = ++seq;
    save.disabled = true;
    status.textContent = t("editor.checking");
    let res;
    try {
      res = await api.post("/questions/preview", { id: qid || newId.value.trim() || "Q_NEW", draft: d, base_id: baseId });
    } catch (e) {
      if (mine === seq) {
        status.textContent = "";
        preview.replaceChildren(errorPanel(e));
        report(e);
      }
      return;
    }
    if (mine !== seq) return;  // a newer edit is already being checked
    save.disabled = !res.valid;
    status.textContent = t(res.valid ? "editor.valid" : "editor.invalid");
    if (!res.valid) {
      preview.replaceChildren(el("div", { class: "error", role: "alert" }, el("strong", {}, t("editor.not_valid")),
        el("ul", {}, res.errors.map(x => el("li", {}, x)))));
      return;
    }
    const graph = el("div");
    const view = select([["concept", t("soal.concept_graph")], ["reified", t("soal.reified_graph")], ["turtle", "Turtle"],
      ["yaml", "YAML"]], "concept");
    const show = async () => {
      if (view.value === "turtle") graph.replaceChildren(code(res.turtle));
      else if (view.value === "yaml") graph.replaceChildren(code(res.yaml));
      else await dot(graph, view.value === "concept" ? res.concept_dot : res.reified_dot);
    };
    view.addEventListener("change", show);
    preview.replaceChildren(el("p", { class: "small muted" }, t("editor.graph_after"), " ", iri(res.graph)),
      field(t("editor.preview_view"), view), graph);
    await show();
  }

  const syncConcepts = () => datalist.replaceChildren(...concepts().map(c => el("option", { value: c })));
  const changed = () => {
    syncConcepts();
    clearTimeout(timer);
    timer = setTimeout(check, 500);
  };
  const triples = rowsEditor([
    { key: "id", label: t("col.id"), kind: "text" },
    { key: "s", label: t("col.subject"), kind: "list", datalist: "concepts-list" },
    { key: "p", label: t("col.predicate"), kind: "select", options: () => predicates.map(p => [p.id, p.id]) },
    { key: "o", label: t("col.object"), kind: "select", options: () => concepts().map(c => [c, c]) },
    { key: "criterion", label: t("col.criterion"), kind: "select", options: () => criteria().map(c => [c, c]) },
    { key: "weight", label: t("col.weight"), kind: "number" },
  ], d.triples, changed);
  const linked = () => {  // the triples' dropdowns list the concepts and criteria
    triples.render();
    changed();
  };
  const criteriaRows = rowsEditor([{ key: "id", label: t("col.id"), kind: "text" },
    { key: "description", label: t("col.description"), kind: "text" }], d.criteria, linked);
  const conceptRows = rowsEditor([{ key: "concept", label: t("col.concept"), kind: "text" },
    { key: "aliases", label: t("col.aliases"), kind: "list" },
    { key: "broader", label: t("col.broader"), kind: "list", datalist: "concepts-list" }], d.concepts, linked);
  const text = key => el("textarea", { rows: 4, value: d[key], oninput: e => { d[key] = e.target.value; changed(); } });
  const language = select([["id", t("editor.lang_id")], ["en", t("editor.lang_en")]], d.language, {
    onchange: async e => {
      d.language = e.target.value;
      predicates = await api.get("/predicates", { lang: d.language });
      triples.render();
      changed();
    } });
  const validated = el("input", { value: d.validated_by || "", oninput: e => { d.validated_by = e.target.value; changed(); } });
  if (newId) newId.addEventListener("input", changed);
  fill(container, datalist,
    newId ? panel(null, el("div", { class: "row" }, field(t("editor.new_id"), newId), field(t("editor.assessment"), target))) : null,
    el("div", { class: "grid2" }, field(t("soal.prompt"), text("prompt")), field(t("soal.reference"), text("reference_answer"))),
    el("div", { class: "row" }, field(t("editor.language"), language), field(t("editor.validated_by"), validated)),
    panel(t("soal.criteria"), criteriaRows.element),
    panel(t("soal.concepts"), el("p", { class: "small muted" }, t("soal.concepts_help")), conceptRows.element),
    panel(t("soal.triples"), el("p", { class: "small muted" }, t("editor.triples_help")), triples.element),
    panel(t("editor.preview"), status, preview),
    el("div", { class: "actions" }, save));
  syncConcepts();
  await check();
}

import { Marked } from "/vendor/marked.esm.js";
import { api, url } from "./api.js";

import ast

pins = {norm(line.split("==")[0]) for line in (root / "requirements.txt").read_text().split() if "==" in line}
modules = set()
for path in [*root.glob("src/**/*.py"), *root.glob("tests/**/*.py"), *root.glob("ontology/*.py")]:
    for node in ast.walk(ast.parse(path.read_text(encoding="utf-8"))):
        if isinstance(node, ast.Import):

            modules.add(node.module.split(".")[0])
dists = packages_distributions()
# tools and optional imports the code does not name directly
needed, todo = set(), [norm(d) for m in modules for d in dists.get(m, [])] + [
    "pytest", "pytest-playwright", "playwright", "uvicorn", "python-multipart", "fastapi", "setuptools", "pip"]
while todo:
    name = todo.pop()
    if name in needed:

    except PackageNotFoundError:

    for r in map(Requirement, reqs):
